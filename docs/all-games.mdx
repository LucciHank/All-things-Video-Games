export const genres = [
  "All",
  "Action",
  "Adventure",
  "RPG",
  "Sports",
  "Strategy",
];

export function GameList() {
  const [selected, setSelected] = useState("All");
  const filtered =
    selected === "All" ? games : games.filter((g) => g.genre === selected);
  return (
    <div>
      <div
        style={{
          display: "flex",
          gap: "0.5rem",
          flexWrap: "wrap",
          marginBottom: "1.5rem",
        }}
      >
        {genres.map((genre) => (
          <button
            key={genre}
            onClick={() => setSelected(genre)}
            style={{
              padding: "0.4rem 1rem",
              borderRadius: "4px",
              border: "1px solid var(--ifm-color-primary)",
              background:
                selected === genre ? "var(--ifm-color-primary)" : "transparent",
              color: selected === genre ? "#fff" : "var(--ifm-color-primary)",
              cursor: "pointer",
              fontWeight: selected === genre ? "bold" : "normal",
            }}
          >
            {genre}
          </button>
        ))}
      </div>
      <ul>
        {filtered.map((game) => (
          <li key={game.title}>
            <strong>{game.title}</strong> — {game.platforms} — {game.year}
            <br />
            {game.description}
          </li>
        ))}
      </ul>
    </div>
  );
}
